1
Server Provisioning & UFW Firewall
Update system repositories and open HTTP (80), HTTPS (443), and SSH (22) in UFW firewall.
sudo apt update && sudo apt upgrade -y
sudo apt install -y git curl wget ufw build-essential openssl
sudo timedatectl set-timezone "Europe/Brussels"

# Configure Firewall
sudo ufw allow OpenSSH
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw --force enable
2
Install Runtime Dependencies (Node 24, PostgreSQL, Redis, Nginx, Certbot)
Install Node.js 24 LTS, PM2 with log rotation, PostgreSQL 15+, Redis Server, Nginx, and Certbot.
# Install Node.js 24 LTS via NodeSource
curl -fsSL https://deb.nodesource.com/setup_24.x | sudo -E bash -
sudo apt install -y nodejs postgresql postgresql-contrib redis-server nginx certbot python3-certbot-nginx

# Setup PM2 & Log Rotation
sudo npm install -g pm2
pm2 install pm2-logrotate
pm2 set pm2-logrotate:max_size 50M
pm2 set pm2-logrotate:compress true
pm2 set pm2-logrotate:retain 14

# Enable Services
sudo systemctl enable postgresql redis-server nginx
sudo systemctl start postgresql redis-server nginx
3
Configure PostgreSQL Database & User
Create PostgreSQL database and user with UTF-8 encoding and schema permissions. 
sudo -u postgres psql -c "CREATE DATABASE ocpp_cms;"
sudo -u postgres psql -c "CREATE USER cms_user WITH PASSWORD 'cms_secure_pass_2026';"
sudo -u postgres psql -c "ALTER ROLE cms_user SET client_encoding TO 'utf8';"
sudo -u postgres psql -c "ALTER ROLE cms_user SET default_transaction_isolation TO 'read committed';"
sudo -u postgres psql -c "ALTER ROLE cms_user SET timezone TO 'Europe/Brussels';"
sudo -u postgres psql -c "GRANT ALL PRIVILEGES ON DATABASE ocpp_cms TO cms_user;"
sudo -u postgres psql -d ocpp_cms -c "GRANT ALL ON SCHEMA public TO cms_user;"

7
Configure Nginx Reverse Proxy (Dual Domain + WebSockets)
Configure Nginx to proxy Frontend on port 3002, REST API on port 3000, OCPP WebSockets on port 9220, and Logs on 3001.
sudo tee /etc/nginx/sites-available/ocpp-cpms.conf > /dev/null <<'EOT'
# 1. Frontend Dashboard
server {
    listen 80;
    server_name ui.mobilitypulse.com;

    location / {
        proxy_pass http://127.0.0.1:3002;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}

# 2. Backend REST API & OCPP WebSockets
server {
    listen 80;
    server_name ocpp.mobilitypulse.com;
    client_max_body_size 50M;

    # REST API
    location /api/ {
        proxy_pass http://127.0.0.1:3000/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Uploaded media assets
    location /uploads/ {
        proxy_pass http://127.0.0.1:3000/uploads/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
    }

    # Real-time Socket.IO Stream
    location /api/realtime/ {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # Live Log Stream WebSocket (Port 3001)
    location /api/ocpp/logs {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }

    # OCPP 1.6-J and 2.0.1/2.1 WebSocket Server (Port 9220)
    location /OCPP/ {
        proxy_pass http://127.0.0.1:9220/OCPP/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }
}
EOT

sudo ln -sf /etc/nginx/sites-available/ocpp-cpms.conf /etc/nginx/sites-enabled/
sudo rm -f /etc/nginx/sites-enabled/default
sudo nginx -t && sudo systemctl restart nginx

⚠️ Clean Reset & Uninstallation
⚠️
Caution: Running these commands will completely wipe the database, delete application files, stop PM2 services, and remove Nginx configurations.
# 1. Stop & Remove PM2 Processes
pm2 delete ocpp-backend 2>/dev/null || true
pm2 delete ocpp-frontend 2>/dev/null || true
pm2 save --force

# 2. Remove Application Directory
sudo rm -rf /var/www/ocpp-cms

# 3. Drop PostgreSQL Database and User
sudo -u postgres psql -c "DROP DATABASE IF EXISTS ocpp_cms;"
sudo -u postgres psql -c "DROP USER IF EXISTS cms_user;"

# 4. Remove Nginx Configurations
sudo rm -f /etc/nginx/sites-available/ocpp-cpms.conf
sudo rm -f /etc/nginx/sites-enabled/ocpp-cpms.conf
sudo systemctl restart nginx
